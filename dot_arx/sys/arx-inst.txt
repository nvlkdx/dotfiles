>| THE GOOD SHIT
###	wipe disks
#	get $DISK
$ fdisk -l|less
#	create container
$ cryptsetup open --type plain --key-file /dev/urandom (--sector-size 4096) /dev/$DISK wipe
#	verify that container exists
$ lsblk
#>	must return crypt block "wipe" as a subset of the disk to be cleared
#	send it
$ dd if=/dev/zero of=/dev/mapper/wipe status=progress bs=1M
#	wait for it to wipe & then check if it worked
$ od /dev/$DISK
#	close the container
$ cryptsetup close wipe
###>	do this for all disks you plan on encrypting

###	make boot partitions on ufd
#	get $BOOT_UFD
$ fdisk -l|less
#	partition
$ fdisk /dev/$BOOT_UFD
#>	TAKE NOTE to do [g] first to create a GPT

#	asterisk means default
--- BEGIN fdisk PARTITION GUIDE TABLE ---
#	128 GB UFD [$BOOT-UFD] "BOOTES":
#	START	SIZE	TYPE	VARIABLE
1	2048	+1G	1	$EFIprt
2	*	*	*	$BOOTprt
--- END fdisk PARTITION GUIDE TABLE ---
#>	[w] to write or [q] to quit

#	format partition 1 ($EFIprt) as FAT32
$ mkfs.fat -F 32 /dev/$EFIprt
##	decide on $ITERTIME in milliseconds
#	encrypt and format partition 2 ($BOOTprt)
$ cryptsetup -M luks2 -c aes-xts-plain64 -s 512 -h sha512 -i $ITERTIME --pbkdf argon2id -y luksFormat /dev/$BOOTprt
$ cryptsetup open /dev/$BOOTprt BOOTES
$ mkfs.ext4 /dev/mapper/BOOTES

###	encrypting the drives
##	decide on a $KEYSIZE [the article im referencing says 4 is ok, though i suspect 8 is better] and $KF_ITERTIME (the $ITERTIME for containers encrypted with keyfiles)
#	create the keyfile
$ dd if=/dev/urandom of=/root/$SSDNYM-key.img bs=512 count=$KEYSIZE iflag=fullblock
#	designate the space for the header
$ truncate -s 16M /root/$SSDNYM-header.img
#	get visual reminder of which drive is which
$ lsblk --nvme
#	encrypt the drive
$ cryptsetup -M luks2 -c aes-xts-plain64 -s 512 -h sha512 -i $KF_ITERTIME --pbkdf argon2id --sector-size 4096 luksFormat /dev/$DRIVE --header /root/$SSDNYM-header.img --key-file /root/$SSDNYM-key.img
#>	[see again the encryption options table on the arch wiki]
#	open the encrypted container
$ cryptsetup open /dev/$DRIVE --header $SSDNYM-header.img --key-file $SSDNYM-key.img $SSDNYM
###>	repeat this section for AXIOMA and ARXIVA in place of $SSDNYM

>| CONTINUING WITH LVM
###	create all necessary lvm abstractions
#	create the physical volumes
$ pvcreate /dev/mapper/AXIOMA
$ pvcreate /dev/mapper/ARXIVA
#	create the volume groups
$ vgcreate AXIOMA /dev/mapper/AXIOMA
$ vgcreate ARXIVA /dev/mapper/ARXIVA
#	create the logical volumes
$ lvcreate -L 128G -n root AXIOMA
$ lvcreate -l 100%FREE -n usr AXIOMA
$ lvcreate -l 100%FREE -n home ARXIVA
#	reduce the size of the root volume to allow for e2scrub later on
$ lvreduce -L -256M AXIOMA/root
