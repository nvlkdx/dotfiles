###	set up swap on zram
#	install the generator
$ pacman -S zram-generator
#	edit the config
$ sudo -e /etc/systemd/zram-generator.conf
--- BEGIN TEXT EDIT BLOCK ---
[zram0]
zram-size = ram / 2
compression-algorithm = zstd
swap-priority = 100
fs-type = swap
--- END TEXT EDIT BLOCK ---
#	get it working
$ reboot

###	set up secure boot
#	check secure boot is in setup mode
$ sbctl status
#>	if it isnt change it to setup mode in the firmware settings
#	create & enroll keys
$ sbctl create-keys
$ sbctl enroll-keys -f
#	check the keys are enrolled
$ sbctl status
#	get $FILE_TO_SIGN
$ sbctl verify
#	get the $KERNEL_IMAGE, which you also need to sign
$ ls /boot/|grep "vmlinuz"
$ sbctl sign -s $FILE_TO_SIGN
$ sbctl sign -s /boot/$KERNEL_IMAGE
#	double check everything that needs to be signed is signed
$ sbctl verify
#	reboot the system and turn secure boot on in the firmware settings
$ reboot
#>	make sure to set up a password for secure boot while youre at the firmware screen

###	set up dnscrypt
#	install
$ pacman -S dnscrypt-proxy
#	edit the config
$ sudo -e /etc/dnscrypt-proxy/dnscrypt-proxy.toml
--- BEGIN TEXT EDIT BLOCK ---
server_names = ['cloudflare', 'cloudflare-ipv6']
listen_addresses = ['127.0.0.1:53', '[::1]:53']
ipv6_servers = true
require_dnssec = true
--- END TEXT EDIT BLOCK ---
#	edit the resolv config
$ sudo -e /etc/resolv.conf
--- BEGIN TEXT EDIT BLOCK ---
nameserver ::1
nameserver 127.0.0.1
options edns0
--- END TEXT EDIT BLOCK ---
#	protect the resolv config from being overwritten
$ chattr +i /etc/resolv.conf
#	double check no programs are using dnscrypt's port
$ ss -lp 'sport = :domain'
#	send it
$ systemctl enable dnscrypt-proxy && systemctl start dnscrypt-proxy
